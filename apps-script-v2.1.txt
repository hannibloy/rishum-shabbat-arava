var SHEET        = 'הרשמות';
var NOTIFY_EMAIL = 'bloyarava1@gmail.com';   // ריק ('') = בלי מייל
var HEADERS = ['נרשם בתאריך', 'שבת פרשת', 'תאריך השבת', 'שם פרטי', 'שם משפחה',
               'טלפון', 'בן/בת', 'סוג אורח', 'ליל שבת', 'יום שבת', 'לינה',
               'הערות', 'השתתפות (₪)', 'סטטוס'];


/* ===== קליטת הרשמה מהאתר ===== */
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    var d = JSON.parse(e.postData.contents);

    // הגנה מכפילויות — אותה הרשמה (אותו clientId) נרשמת פעם אחת בלבד
    var cache = CacheService.getScriptCache();
    var key = d.clientId ? 'reg_' + d.clientId : '';
    if (key && cache.get(key)) return out({ ok: true, duplicate: true });

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sh = ss.getSheetByName(SHEET) || ss.insertSheet(SHEET);

    // בפעם הראשונה — בונה את הכותרות
    if (sh.getLastRow() === 0) {
      sh.appendRow(HEADERS);
      sh.setFrozenRows(1);
      sh.getRange(1, 1, 1, HEADERS.length)
        .setFontWeight('bold')
        .setBackground('#F5EDDF');
      sh.setRightToLeft(true);
    }

    sh.appendRow([
      new Date(),
      d.parasha    || '',
      d.shabbatDate|| '',
      d.firstName  || '',
      d.lastName   || '',
      "'" + (d.phone || ''),          // הגרש שומר על האפס בהתחלה
      d.gender     || '',
      d.guestType  || '',
      d.mealFri ? '✓' : '',
      d.mealSat ? '✓' : '',
      d.sleeping === 'yes' ? 'לן/ה' : 'רק סעודות',
      d.notes      || '',
      Number(d.amount) || 0,
      Number(d.amount) > 0 ? 'בחר/ה להשתתף' : 'ללא'
    ]);

    if (key) cache.put(key, '1', 21600);   // 6 שעות

    try { updateSummary(ss); } catch (err) {}   // דוח המטבח לא יפיל הרשמה
    try { if (NOTIFY_EMAIL) sendMail(d); } catch (err) {}

    return out({ ok: true });

  } catch (err) {
    return out({ ok: false, error: String(err) });
  } finally {
    lock.releaseLock();
  }
}


/* ===== מייל גיבוי אלייך ===== */
function sendMail(d) {
  var meals = [];
  if (d.mealFri) meals.push('ליל שבת');
  if (d.mealSat) meals.push('יום שבת');
  var phone = String(d.phone || '').replace(/\D/g, '').replace(/^0/, '972');
  var body =
    'שם: ' + (d.firstName || '') + ' ' + (d.lastName || '') + '\n' +
    'טלפון: ' + (d.phone || '') + '\n' +
    'שבת: ' + (d.parasha || '') + ' (' + (d.shabbatDate || '') + ')\n' +
    'סעודות: ' + (meals.length ? meals.join(' + ') : 'ללא') + '\n' +
    'לינה: ' + (d.sleeping === 'yes' ? 'כן' : 'לא') + '\n' +
    'השתתפות: ' + (Number(d.amount) || 0) + ' ₪\n' +
    'הערות: ' + (d.notes || '') + '\n\n' +
    'וואטסאפ: https://wa.me/' + phone;
  MailApp.sendEmail(NOTIFY_EMAIL,
    'הרשמה לשבת: ' + (d.firstName || '') + ' ' + (d.lastName || '') + ' — ' + (d.parasha || ''),
    body);
}


/* ===== לשונית "השבת הקרובה" — דוח המטבח האוטומטי ===== */
function updateSummary(ss) {
  var sh  = ss.getSheetByName(SHEET);
  var sum = ss.getSheetByName('השבת הקרובה') || ss.insertSheet('השבת הקרובה');

  sum.clear();
  sum.setRightToLeft(true);

  var data = sh.getDataRange().getValues();
  if (data.length < 2) return;

  var last = data[data.length - 1][1];   // הפרשה של ההרשמה האחרונה
  var total = 0, boys = 0, girls = 0, sleep = 0,
      fri = 0, sat = 0, paid = 0, sum_amount = 0;

  for (var i = 1; i < data.length; i++) {
    if (data[i][1] !== last) continue;
    total++;
    if (data[i][6] === 'בן')  boys++;
    if (data[i][6] === 'בת')  girls++;
    if (data[i][10] === 'לן/ה') sleep++;
    if (data[i][8] === '✓') fri++;
    if (data[i][9] === '✓') sat++;
    if (Number(data[i][12]) > 0) { paid++; sum_amount += Number(data[i][12]); }
  }

  var rows = [
    ['שבת פרשת ' + last, ''],
    ['', ''],
    ['סה"כ נרשמו',            total],
    ['בנים / בנות',           boys + ' / ' + girls],
    ['לנים במתחם',            sleep],
    ['', ''],
    ['מנות — סעודת ליל שבת',  fri],
    ['מנות — סעודת יום שבת',  sat],
    ['', ''],
    ['בחרו להשתתף בעלויות',   paid + ' מתוך ' + total],
    ['סה"כ השתתפות',          sum_amount + ' ₪']
  ];

  sum.getRange(1, 1, rows.length, 2).setValues(rows);
  sum.getRange(1, 1).setFontWeight('bold').setFontSize(15);
  sum.getRange(2, 1, rows.length - 1, 1).setFontWeight('bold');
  sum.setColumnWidth(1, 230);
  sum.setColumnWidth(2, 130);
}


function out(o) {
  return ContentService
    .createTextOutput(JSON.stringify(o))
    .setMimeType(ContentService.MimeType.JSON);
}
